10/// Enum representing the result size of a value/expression/register/etc in bytes.
11/// just friendly names really
12#[allow(non_camel_case_types)]
13#[derive(Debug, PartialOrd, PartialEq, Ord, Eq, Hash, Clone, Copy)]
14pub enum Size {
15    BYTE = 1,
16    B_2 = 2,
17    B_4 = 4,
18    B_6 = 6,
19    B_8 = 8,
20    B_10 = 10,
21    B_16 = 16,
22    B_32 = 32,
23    B_64 = 64,
24}

26impl Size {
27    pub fn in_bytes(self) -> u8 {
28        self as u8
29    }

31    pub fn as_literal(self) -> syn::Ident {
32        syn::Ident::new(match self {
33            Size::BYTE  => "i8",
34            Size::B_2  => "i16",
35            Size::B_4 => "i32",
36            Size::B_6 => "i48",
37            Size::B_8 => "i64",
38            Size::B_10 => "i80",
39            Size::B_16 => "i128",
40            Size::B_32 => "i256",
41            Size::B_64 => "i512",
42        }, Span::mixed_site())
43    }
44}

46/// AST type for parsed dynasm label reference syntax.
47/// Represents a reference to somewhere, either a label or an address, with an optional expression offset.
48#[derive(Debug, Clone)]
49pub struct JumpTarget {
50    pub kind: JumpTargetKind,
51    pub offset: Option<syn::Expr>
52}
53
54/// The different types of jump targets
55#[derive(Debug, Clone)]
56pub enum JumpTargetKind {
57    // note: these symbol choices try to avoid stuff that is a valid starting symbol for parse_expr
58    // in order to allow the full range of expressions to be used. the only currently existing ambiguity is
59    // with the symbol <, as this symbol is also the starting symbol for the universal calling syntax <Type as Trait>.method(args)
60
61    /// A global label: `->label (+/- offset_expr)`
62    Global(syn::Ident),
63    /// A backwards local label: `<label (+/- offset_expr)`
64    Backward(syn::Ident),
65    /// A forwards local label: `<label (+/- offset_expr)`
66    Forward(syn::Ident),
67    /// A dynamic label: `=>label_expr (+/- offset_expr)`
68    Dynamic(TokenTree),
69    /// A reference to an absolute address: `extern address_expr (+/- offset_expr)`
70    Absolute(TokenTree),
71    /// A relative offset. This doesn't have dedicated syntax but used internally by x86 which needs them for rip-relative addressing
72    Relative(TokenTree),
73}
74
75impl ParseOpt for JumpTarget {
76    fn parse(input: parse::ParseStream) -> parse::Result<Option<JumpTarget>> {
77        // extern label
78        if eat_pseudo_keyword(input, "extern") {
79            let expr: syn::Expr = input.parse()?;
80
81            return Ok(Some(JumpTarget { kind: JumpTargetKind::Absolute(delimited(expr)), offset: None }));
82        }
83
84        // -> global_label
85        let kind = if input.peek(Token![->]) {
86            let _: Token![->] = input.parse()?;
87            let name: syn::Ident = input.parse()?;
88
89            JumpTargetKind::Global(name)
90
91        // > forward_label
92        } else if input.peek(Token![>]) {
93            let _: Token![>] = input.parse()?;
94            let name: syn::Ident = input.parse()?;
95
96            JumpTargetKind::Forward(name)
97
98        // < backwards_label
99        } else if input.peek(Token![<]) {
100            let _: Token![<] = input.parse()?;
101            let name: syn::Ident = input.parse()?;
102
103            JumpTargetKind::Backward(name)
104
105        // => dynamic_label
106        } else if input.peek(Token![=>]) {
107            let _: Token![=>] = input.parse()?;
108
109            let expr: syn::Expr = if input.peek(syn::token::Paren) {
110                let inner;
111                let _ = syn::parenthesized!(inner in input);
112                let inner = &inner;
113
114                inner.parse()?
115            } else {
116                input.parse()?
117            };
118
119            JumpTargetKind::Dynamic(delimited(expr))
120
121        // nothing
122        } else {
123            return Ok(None);
124        };
125
126        // parse optional offset
127        let offset = if input.peek(Token![-]) || input.peek(Token![+]) {
128            if input.peek(Token![+]) {
129                let _: Token![+] = input.parse()?;
130            }
131
132            let expr: syn::Expr = input.parse()?;
133            Some(expr)
134
135        } else {
136            None
137        };
138
139        Ok(Some(JumpTarget::new(kind, offset)))
140    }
141}

143impl JumpTarget {
144    pub fn new(kind: JumpTargetKind, offset: Option<syn::Expr>) -> JumpTarget {
145        JumpTarget {
146            kind,
147            offset
148        }
149    }
150
151    pub fn target_is_absolute(&self) -> bool {
152        match self.kind {
153            JumpTargetKind::Absolute(_) => true,
154            _ => false
155        }
156    }
157
158    /// Takes a jump and encodes it as a relocation starting `start_offset` bytes ago, relative to `ref_offset`.
159    /// Any data detailing the type of relocation emitted should be contained in `data`, which is emitted as a tuple of u8's.
160    pub fn encode(self, field_offset: u8, ref_offset: u8, relative_encoding: bool, encoding: RelocationEncoding) -> Stmt {
161        let kind = match (relative_encoding, self.target_is_absolute()) {
162            (true, false) => RelocationKind::Relative,
163            (true, true) => RelocationKind::RelToAbs,
164            (false, false) => RelocationKind::AbsToRel,
165            (false, true) => RelocationKind::Absolute,
166        };
167
168        let target_offset = if let Some(offset) = self.offset {
169            delimited(offset)
170        } else {
171            TokenTree::Literal(Literal::isize_suffixed(0))
172        };
173
174        // Create a relocation descriptor, containing all information about the actual jump except for the target itself.
175        let relocation = Relocation {
176            target_offset,
177            field_offset,
178            ref_offset,
179            kind,
180            encoding
181        };
182        match self.kind {
183            JumpTargetKind::Global(ident) => Stmt::GlobalJumpTarget(ident, relocation),
184            JumpTargetKind::Backward(ident) => Stmt::BackwardJumpTarget(ident, relocation),
185            JumpTargetKind::Forward(ident) => Stmt::ForwardJumpTarget(ident, relocation),
186            JumpTargetKind::Dynamic(expr) => Stmt::DynamicJumpTarget(expr, relocation),
187            JumpTargetKind::Absolute(expr)
188            | JumpTargetKind::Relative(expr) => Stmt::ValueJumpTarget(expr, relocation),
189        }
190    }
191
192    pub fn span(&self) -> Span {
193        match &self.kind {
194            JumpTargetKind::Global(ident)
195            | JumpTargetKind::Backward(ident)
196            | JumpTargetKind::Forward(ident) => ident.span(),
197            JumpTargetKind::Dynamic(expr)
198            | JumpTargetKind::Absolute(expr)
199            | JumpTargetKind::Relative(expr) => expr.span(),
200        }
201    }
202}

205/// Different relocation behaviours for an encoding-target pair
206/// This specifies how the relocation has to be adapted when the assembling buffer is moved.
207/// Note that there's no absolute kind as an absolute encoding to an absolute target
208/// is just an immediate.
209#[derive(Debug, Clone, Copy)]
210#[repr(u8)]
211pub enum RelocationKind {
212    /// A relative encoding to a relative target
213    Relative = 0,
214    /// An absolute encoding to a relative target
215    AbsToRel = 1,
216    /// A relative encoding to an absolute target
217    RelToAbs = 2,
218    /// An absolute encoding to an absolute target (idk why)
219    Absolute = 3,
220}

223/// Specifies the way that a relocation is encoded.
224#[derive(Debug, Clone, Copy)]
225pub enum RelocationEncoding {
226    /// Just as a value of a certain size, encoded little-endian
227    Simple(Size),
228    /// An encoding custom to a certain instruction set. value 0-59 are available here.
229    Custom(u8)
230}
231
232impl RelocationEncoding {
233    /// pack the type of relocation that this is, based on the `kind` and `encoding` fields, in a
234    /// single byte. This should match with the decoding logic in `dynasmrt`.
235    pub fn encode(&self, kind: RelocationKind) -> u8 {
236        (match self {
237            RelocationEncoding::Simple(size) => match size {
238                Size::BYTE => 0,
239                Size::B_2 => 1,
240                Size::B_4 => 2,
241                Size::B_8 => 3,
242                _ => panic!("Unencodable size given for simple relocation")
243            },
244            RelocationEncoding::Custom(code) => {
245                (code + 4) & 0x3F
246            }
247        }) | ((kind as u8) << 6)
248    }
249}

262/// An abstract representation of a dynasm runtime statement to be emitted
263#[derive(Debug, Clone)]
264pub enum Stmt {
265    // simply push data into the instruction stream. unsigned
266    Const(u64, Size),
267    // push data that is stored inside of an expression. unsigned
268    ExprUnsigned(TokenTree, Size),
269    // push signed data into the instruction stream. signed
270    ExprSigned(TokenTree, Size),
271
272    // extend the instruction stream with unsigned bytes
273    Extend(Vec<u8>),
274    // extend the instruction stream with unsigned bytes
275    ExprExtend(TokenTree),
276    // align the instruction stream to some alignment
277    Align(TokenTree, TokenTree),
278
279    // label declarations
280    GlobalLabel(syn::Ident),
281    LocalLabel(syn::Ident),
282    DynamicLabel(TokenTree),
283
284    // and their respective relocations (as expressions as they differ per assembler).
285    GlobalJumpTarget(syn::Ident, Relocation),
286    ForwardJumpTarget(syn::Ident, Relocation),
287    BackwardJumpTarget(syn::Ident, Relocation),
288    DynamicJumpTarget(TokenTree, Relocation),
289    ValueJumpTarget(TokenTree, Relocation),
290
291    // a random statement that has to be inserted between assembly hunks
292    Stmt(TokenStream)
293}

317/// Takes an arbitrary tokenstream as input, and ensures it can be interpolated safely.
318/// returns a tokentree representing either a single token, or a delimited group.
319///
320/// If the given tokenstream contains multiple tokens, it will be parenthesized.
321///
322/// this will panic if given an empty tokenstream.
323/// this would use delimiter::None if not for https://github.com/rust-lang/rust/issues/67062
324pub fn delimited<T: ToTokens>(expr: T) -> TokenTree {
325    let stream = expr.into_token_stream();
326
327    // the stream api is very limited, but cloning a stream is luckily cheap.
328    // so to check how many tokens are contained we can do this.
329    let mut iter = stream.clone().into_iter();
330    let first = iter.next().unwrap();
331    if iter.next().is_none() {
332        return first;
333    }
334
335    let span = stream.span();
336    let mut group = Group::new(
337        proc_macro2::Delimiter::Parenthesis, stream
338    );
339    group.set_span(span);
340    TokenTree::Group(group)
341}

343/// Checks if the given `Group` is a parenthesized expression to work around rustc giving
344/// Unnecessary parenthesis warnings in macro-generated code, if this tokentree were to be used
345/// as the argument to a single argument function
346///
347/// i.e. `function(#arg)` expanding to `function((expr))`, which should instead be expanded to
348/// `function(expr)`
349///
350/// To check if this is valid, we should a: test that this tokentree node is a parenthesis delimited
351/// node and b: there are no commas in its internal tokentree, because then it'd be a tuple, and
352/// this transform would be invalid
353pub fn is_parenthesized(group: &Group) -> bool {
354    if group.delimiter() != Delimiter::Parenthesis {
355        return false
356    }
357
358    for item in group.stream() {
359        if let TokenTree::Punct(punct) = item {
360            if punct.as_char() == ',' {
361                return false
362            }
363        }
364    }
365
366    true
367}
368
369/// Returns the given `TokenTree`, but if it's a parenthesized group, it will change this
370/// to a None-delimited group, if `is_parenthesized` deems this to be a valid transform
371///
372/// this is intended to work around unneeded parenthesis around function arguments warnings
373pub fn strip_parenthesis(expr: &mut TokenTree) {
374    if let TokenTree::Group(group) = &*expr {
375        if is_parenthesized(group) {
376            let mut stripped = TokenTree::Group(Group::new(Delimiter::None, group.stream()));
377            stripped.set_span(group.span());
378            *expr = stripped;
379        }
380    }
381}

383/// Create a bitmask with `scale` bits set
384pub fn bitmask(scale: u8) -> u32 {
385    1u32.checked_shl(u32::from(scale)).unwrap_or(0).wrapping_sub(1)
386}
